function doGet() {
  return HtmlService.createHtmlOutputFromFile('Index')
      .setTitle('Rekap Presensi & Izin Guru')
      .addMetaTag('viewport', 'width=device-width, initial-scale=1')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function getSheet(sheetName) {
  return SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName);
}

function getDaftarGuru() {
  const sheetGuru = getSheet("Daftar Guru");
  if (!sheetGuru) return []; 
  
  const data = sheetGuru.getDataRange().getValues();
  let listGuru = [];
  
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] !== "") { 
      listGuru.push(data[i][0]);
    }
  }
  return listGuru.sort(); 
}

function submitIzin(nama, kategori) {
  if (!nama || !kategori) return getTodayData(); 
  
  const sheet = getSheet("Data Izin");
  const timestamp = new Date();
  const dateStr = Utilities.formatDate(timestamp, "GMT+7", "yyyy-MM-dd"); 
  
  sheet.appendRow([timestamp, dateStr, nama, kategori]);
  return getTodayData(); 
}

function getTodayData() {
  const sheet = getSheet("Data Izin");
  const data = sheet.getDataRange().getValues();
  const todayStr = Utilities.formatDate(new Date(), "GMT+7", "yyyy-MM-dd");
  const todayDisplay = Utilities.formatDate(new Date(), "GMT+7", "EEEE, d MMMM yyyy");

  let results = {
    "Tugas kedinasan": [],
    "Sakit": [],
    "Izin terlambat": [],
    "Izin keluar saat jam dinas (keperluan penting, dan kembali ke sekolah)": [],
    "Izin pulang sebelum waktunya": [],
    "Keperluan pribadi": [],
    "Cuti": []
  };

  let guruIzin = []; 

  for (let i = 1; i < data.length; i++) {
    let rawTimestamp = data[i][0];
    let rowDate = data[i][1]; 
    let rowDateStr = "";
    
    if (rowDate instanceof Date) {
      rowDateStr = Utilities.formatDate(rowDate, "GMT+7", "yyyy-MM-dd");
    } else {
      rowDateStr = String(rowDate).trim();
    }

    if (rowDateStr === todayStr) {
      let nama = data[i][2];
      let kat = data[i][3];
      
      // Ambil data jam dari Kolom A (Timestamp)
      let jamStr = "";
      if (rawTimestamp instanceof Date) {
        jamStr = Utilities.formatDate(rawTimestamp, "GMT+7", "HH:mm");
      }
      
      if (results[kat]) {
        // Simpan nama beserta jamnya
        results[kat].push({ nama: nama, jam: jamStr });
        guruIzin.push(nama);
      }
    }
  }

  const daftarGuruUtuh = getDaftarGuru();
  let guruHadir = daftarGuruUtuh.filter(nama => !guruIzin.includes(nama));

  return {
    tanggal: todayDisplay,
    rekap: results,
    hadir: guruHadir,
    totalGuru: daftarGuruUtuh.length, 
    daftarGuru: daftarGuruUtuh 
  };
}
