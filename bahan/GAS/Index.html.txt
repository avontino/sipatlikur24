<script>
  document.addEventListener("DOMContentLoaded", function() {
    loadData();
  });

  function loadData() {
    google.script.run
      .withSuccessHandler(renderData)
      .withFailureHandler(function(err) { alert("Gagal memuat data: " + err); })
      .getTodayData();
  }

  function kirimData() {
    const nama = document.getElementById("namaGuru").value;
    const kategori = document.getElementById("kategoriIzin").value;
    
    if (nama === "" || kategori === "") {
      alert("Mohon pilih Nama Anda dan Keterangan!");
      return;
    }

    const btn = document.getElementById("btnKirim");
    btn.innerText = "Mengirim...";
    btn.disabled = true;

    google.script.run
      .withSuccessHandler(function(data) {
        document.getElementById("namaGuru").selectedIndex = 0;
        document.getElementById("kategoriIzin").selectedIndex = 0;
        btn.innerText = "Kirim";
        btn.disabled = false;
        renderData(data);
      })
      .withFailureHandler(function(err) {
        alert("Gagal mengirim data: " + err);
        btn.innerText = "Kirim";
        btn.disabled = false;
      })
      .submitIzin(nama, kategori);
  }

  function buatItemPoll(judul, daftarNama, totalGuru, warnaClass, idUnik) {
    const count = daftarNama.length;
    const persentase = totalGuru > 0 ? (count / totalGuru) * 100 : 0;
    
    let userListHtml = "";
    if (count > 0) {
      userListHtml = `<div class="user-list" id="${idUnik}">`;
      daftarNama.forEach(item => {
        // Cek apakah item adalah string biasa (untuk "Hadir") atau object (untuk "Izin" yang ada jamnya)
        if (typeof item === 'object') {
          userListHtml += `<div class="user-item">
                             <span>👤 ${item.nama}</span>
                             <span class="time-badge">${item.jam} WIB</span>
                           </div>`;
        } else {
          userListHtml += `<div class="user-item">
                             <span>👤 ${item}</span>
                           </div>`;
        }
      });
      userListHtml += `</div>`;
    }

    return `
      <div class="poll-item">
        <div class="poll-header" onclick="toggleList('${idUnik}')">
          <div class="poll-title">${judul}</div>
          <div class="poll-count">${count}</div>
        </div>
        <div class="poll-bar-bg">
          <div class="poll-bar-fill ${warnaClass}" style="width: ${persentase}%"></div>
        </div>
        ${userListHtml}
      </div>
    `;
  }

  function renderData(data) {
    const selectGuru = document.getElementById("namaGuru");
    selectGuru.innerHTML = '<option value="" disabled selected>Pilih Nama Anda...</option>';
    
    if (data.daftarGuru && data.daftarGuru.length > 0) {
      data.daftarGuru.forEach(guru => {
        const option = document.createElement("option");
        option.value = guru;
        option.text = guru;
        selectGuru.appendChild(option);
      });
    }

    const hari = {"Sunday": "Minggu", "Monday": "Senin", "Tuesday": "Selasa", "Wednesday": "Rabu", "Thursday": "Kamis", "Friday": "Jumat", "Saturday": "Sabtu"};
    let tanggalIndo = data.tanggal.replace(/[a-zA-Z]+/, match => hari[match] || match);
    document.getElementById("tanggalText").innerText = "Rekap Hari Ini: " + tanggalIndo;
    
    const container = document.getElementById("pollResults");
    let htmlContent = "";
    const totalGuru = data.totalGuru;

    htmlContent += buatItemPoll("Hadir (Di Sekolah)", data.hadir, totalGuru, "fill-blue", "list-hadir");

    for (const [kategori, users] of Object.entries(data.rekap)) {
      let idUnik = "list-" + kategori.replace(/[^a-zA-Z0-9]/g, '');
      htmlContent += buatItemPoll(kategori, users, totalGuru, "fill-green", idUnik);
    }

    container.innerHTML = htmlContent;
  }

  function toggleList(id) {
    const list = document.getElementById(id);
    if (list) {
      list.style.display = (list.style.display === "block") ? "none" : "block";
    }
  }
